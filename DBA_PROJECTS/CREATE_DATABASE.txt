#==================================================
# CREACIÓN DE DB
#==================================================
#**************************************************
# 1) CONTEXTO Y ENTORNO
#**************************************************
# Vamos a crear una base de datos Oracle en una máquina virtual
# Linux Server v8.10. Los recursos que le daremos a dicha VM serán
# los siguientes:
# - 3052 MB de memoria RAM.
# - 3 CPUs.
# - 200 GB de disco.
# Hemos instalado Oracle Database 19c en la máquina siguiendo el
# proceso oficial de instalación. Una vez la tenemos preparada,
# procedemos a la creación de la base de datos.
# Para acceder a la VM, hemos configurado una regla NAT que redirige
# el puerto 2222 del host al puerto 22 de la máquina virtual. Por
# ello, ejecutaremos la siguiente conexión:
ssh -p 2222 oracle@localhost
#**************************************************
# 2) CREACIÓN DE UNA INSTANCIA MANUAL [1]
#**************************************************
# Para crear la instancia de forma manual, usamos el comando
# "SQL> CREATE DATABASE".
# Vamos a ver su alcance. En este caso, no vamos a configurar ASM.
# Para ver el uso de DBCA, consultamos [8] (modo no interactivo)
# y [9] (modo interactivo).
#--------------------------------------------------
# 2.1 ) ESTABLECER ORACLE_SID
#--------------------------------------------------
# El primer paso es elegir un nombre para la variable ORACLE_SID.
# En este caso, la usaremos como ORCLCDB.
# Vamos a crear un directorio ENV dentro del home del usuario oracle,
# y en él añadiremos los archivos de entorno para cada base de datos
# que creemos.
mkdir -p ~/ENV
vi ~/ENV/ORCLCDB.env
 # DB PARAMS
 export ORACLE_SID=ORCLCDB

 # OTHERS
 export PS1="\u[$ORACLE_SID] \w>"
#
#--------------------------------------------------
# 2.2 ) MÉTODO DE AUTENTICACIÓN [2]
#--------------------------------------------------
# El siguiente paso trata sobre los métodos de autenticación.
# En este caso, siempre podremos entrar desde el sistema operativo
# como usuario oracle, que ya tiene privilegios de SYSDBA, pero
# igualmente crearemos un fichero de contraseñas para completar la
# configuración.
# Utilizaremos la utilidad ORAPWD. Este comando crea y mantiene un
# archivo de contraseñas.
orapwd file=$ORACLE_HOME/dbs/orapw$ORACLE_SID \
    FORCE=y \
    FORMAT=12.2 \
    SYS=password
#
#--------------------------------------------------
# 2.3 ) CREACIÓN DEL PFILE
#--------------------------------------------------
# Estamos listos para crear el PFILE de la base de datos. Lo hacemos
# porque aún no podemos generar un SPFILE, ya que la instancia aún
# no existe.
# El fichero estará en $ORACLE_HOME/dbs/init$ORACLE_SID.ora.
vi $ORACLE_HOME/dbs/initORCLCDB.ora

 db_name='ORCLCDB'
 memory_target=1G
 processes = 150
 db_block_size=8192
 db_domain=''
 db_recovery_file_dest='/u03/fast_recovery_area'
 db_recovery_file_dest_size=30G
 diagnostic_dest='/u01/app/oracle'
 dispatchers='(PROTOCOL=TCP) (SERVICE=SRVORCLCDB)'
 open_cursors=300
 enable_pluggable_database=true
 remote_login_passwordfile='EXCLUSIVE'
 undo_tablespace='UNDOTBS1'
 # You may want to ensure that control files are created on separate physical
 # devices
 control_files = (/u01/ctrlf/control01.ctl, /u02/ctrlf/control02.ctl)
 compatible ='12.0.0'
#
# Este PFILE es mínimo, pero funcional.
# En el PFILE, la gestión del UNDO es automática por defecto,
# aunque podemos hacerla explícita añadiendo UNDO_MANAGEMENT=AUTO.
# Con este modo, el UNDO se guarda en el tablespace de UNDO y lo
# gestiona el motor de la base de datos.
#
# Si añadimos alguno de estos parámetros al PFILE:
# DB_CREATE_FILE_DEST, DB_CREATE_ONLINE_LOG_DEST_n o
# DB_RECOVERY_FILE_DEST, indicamos a la base de datos que cree y
# gestione los ficheros asociados a esas estructuras.
# Oracle recomienda usar Fast Recovery Area.
#--------------------------------------------------
# 2.4 ) CONEXIÓN A LA INSTANCIA Y SPFILE
#--------------------------------------------------
# Con las variables de entorno correctamente asignadas, entramos como
# usuario oracle y conectamos a la instancia en estado idle.
su - oracle
source ~/ENV/ORCLCDB.env
sqlplus /nolog
SQL> CONNECT / AS SYSDBA

 Connected to an idle instance.
#
# Creamos un SPFILE con el PFILE que teníamos.
# Se crea en la ruta por defecto "$ORACLE_HOME/dbs/spfileORCLCDB.ora".
#
SQL> CREATE SPFILE FROM PFILE='/u01/app/oracle/product/19.0.0/dbhome_1/dbs/initORCLCDB.ora';

 File created.
#
# Existen dos tipos de parámetros de inicio:
# - Dinámicos: pueden cambiarse en la propia instancia y se aplican
#   de forma inmediata.
# - Estáticos: requieren reinicio de la base de datos.
# Para cambiarlos se usa ALTER SYSTEM SET, y este comando admite varios
# alcances o scopes.
# - SPFILE: cambia el valor solo en el SPFILE y no altera la instancia actual.
# - MEMORY: cambia el parámetro en memoria para esta instancia, pero no
#   persiste en futuros reinicios.
# - BOTH: combina las dos anteriores. Solo sirve para parámetros dinámicos.
# Si usamos SPFILE o BOTH y la base de datos se inicializa con un archivo
# SPFILE, SCOPE=BOTH será el comportamiento por defecto.
# Si hemos usado un PFILE, el comportamiento por defecto será MEMORY.
# También podemos usar ALTER SYSTEM RESET para dejar el parámetro con
# su valor por defecto.
# CUIDADO: el comportamiento de este comando cambia según la base de datos
# sea CDB o PDB.
# Si usamos RMAN, podemos hacer un backup solo del SPFILE, aunque si
# hacemos un backup completo de la base de datos, este ya quedará incluido.
# Algunos métodos para ver los parámetros son:
SQL> SHOW PARAMETERS --Sesión actual
SQL> SHOW SPPARAMETERS --Los del SPFILE
SQL> CREATE PFILE FROM SPFILE; --Genera un PFILE a partir del SPFILE actual
SQL> SELECT * FROM V$PARAMETER2; --Sesión actual
SQL> SELECT * FROM V$SYSTEM_PARAMETER2; --Instancia actual
SQL> SELECT * FROM V$SPPARAMETER; --Contenido actual del SPFILE
#
SQL> SELECT INSTANCE_NAME,DATABASE_STATUS,STATUS FROM V$INSTANCE;

 INSTANCE_NAME    DATABASE_STATUS   STATUS
 ---------------- ----------------- ------------
 ORCLCDB          ACTIVE            STARTED

# En este punto, la memoria ya está asignada y los procesos han sido
# levantados, pero la base de datos todavía no existe.
#--------------------------------------------------
# 2.5 ) CREACIÓN DE LA DB
#--------------------------------------------------
# Antes de ejecutar CREATE DATABASE, es importante comprobar que los
# directorios de trabajo y de control existan. Oracle no los crea solo.
mkdir -p /u01/ctrlf /u02/ctrlf /u03/online_redo /u02/oradata/ORCLCDB

CREATE DATABASE ORCLCDB
   USER SYS IDENTIFIED BY "0R4C|34DM1N"
   USER SYSTEM IDENTIFIED BY "ORASYSTEM"
   LOGFILE GROUP 1 ('/u03/online_redo/redo01a.log') SIZE 100M BLOCKSIZE 512,
           GROUP 2 ('/u03/online_redo/redo02a.log') SIZE 100M BLOCKSIZE 512,
           GROUP 3 ('/u03/online_redo/redo03a.log') SIZE 100M BLOCKSIZE 512
   MAXLOGHISTORY 100
   MAXLOGFILES 16
   MAXLOGMEMBERS 3
   MAXDATAFILES 1024
   CHARACTER SET AL32UTF8
   NATIONAL CHARACTER SET AL16UTF16
   EXTENT MANAGEMENT LOCAL
   DATAFILE '/u02/oradata/ORCLCDB/system01.dbf'
     SIZE 700M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE 15G
   SYSAUX DATAFILE '/u02/oradata/ORCLCDB/sysaux01.dbf'
     SIZE 550M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE 15G
   DEFAULT TABLESPACE USERS
      DATAFILE '/u02/oradata/ORCLCDB/users01.dbf'
      SIZE 500M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE 10G
   DEFAULT TEMPORARY TABLESPACE TEMPTS1
      TEMPFILE '/u02/oradata/ORCLCDB/temp01.dbf'
      SIZE 200M REUSE AUTOEXTEND ON NEXT 640K MAXSIZE 15G
   UNDO TABLESPACE UNDOTBS1
      DATAFILE '/u02/oradata/ORCLCDB/undotbs01.dbf'
      SIZE 200M REUSE AUTOEXTEND ON NEXT 5120K MAXSIZE 15G
    ENABLE PLUGGABLE DATABASE
      SEED
      FILE_NAME_CONVERT =
      ('/u02/oradata/ORCLCDB/',
       '/u02/oradata/ORCLCDB/pdbseed/');

  Database created.

# CUIDADO: las contraseñas son case-sensitive.
#
# CUIDADO: este comando NO crea directorios; deben existir previamente.
# Si la ejecución falla, habrá que limpiar todo lo que haya dejado la
# anterior ejecución.
#
# CUIDADO: si vamos a usar OMF, hay que revisar los parámetros por
# defecto del CREATE DATABASE.
#
# NOTA: EXTENT MANAGEMENT LOCAL crea un bitmap en cada datafile para
# gestionar los extents. Esta es la opción recomendada, aunque NO es
# la opción por defecto, por lo que hay que especificarla.
#
# NOTA: el tablespace SYSAUX es un tablespace auxiliar de SYSTEM,
# obligatorio y que sirve como área por defecto para muchas
# funcionalidades y productos que antes requerían de uno propio.
# También reduce la carga sobre el tablespace de SYSTEM. Podemos usar
# V$SYSAUX_OCCUPANTS para ver los componentes que lo ocupan [4].
#
# NOTA: en el CREATE DATABASE podemos elegir si los tablespaces por
# defecto serán bigfile o smallfile. Podemos consultar el tipo actual
# con:
SQL> SELECT PROPERTY_VALUE FROM DATABASE_PROPERTIES
WHERE PROPERTY_NAME = 'DEFAULT_TBS_TYPE';
# Por defecto, el tipo es smallfile.
# También podemos alterar este comportamiento especificando nosotros
# la opción, salvo en los tablespaces de SYSTEM y SYSAUX.
#
# NOTA: si tenemos UNDO_MANAGEMENT en AUTO y no especificamos un
# tablespace de UNDO, Oracle creará uno por defecto llamado SYS_UNDOTBS.
#
# NOTA: Oracle recomienda crear un tablespace por defecto para los
# usuarios no-SYSTEM. Si no lo hacemos, el tablespace por defecto será
# SYSTEM. Lo mismo ocurre con el tablespace temporal. Si no creamos
# uno por defecto y no especificamos uno al crear el usuario, el
# tablespace temporal por defecto será SYSTEM (si no está en locally
# managed [5]), lo que es una mala práctica.
# De igual forma, si no creamos ningún tablespace de UNDO y el
# parámetro UNDO_TABLESPACE no está en AUTO, los datos de UNDO se
# escriben en SYSTEM.
#
# Cuando ejecutamos un "CREATE DATABASE" hacemos al menos estas
# operaciones:
# - Crear los datafiles, controlfiles y online redologs, y establecer
#   el modo ARCHIVELOG.
# - Crear los tablespaces de SYSTEM y SYSAUX.
# - Crear el diccionario de datos.
# - Establecer el charset y la zona horaria.
#   La zona horaria por defecto es la del host y puede cambiarse para
#   ciertas sesiones. Para ver las zonas horarias válidas, revisa:
#   "$ORACLE_HOME/oracore/zoneinfo/timezlrg_11.dat" (el que se usa por
#   defecto)
#   "$ORACLE_HOME/oracore/zoneinfo/timezone_11.dat" (contiene menos zonas)
#   o en SQL*Plus ejecuta:
SQL> SELECT * FROM V$TIMEZONE_NAMES;
# - Montar y abrir la base de datos.
#
# NOTA: para consultar las trazas, ejecuta:
SQL> SELECT * FROM V$DIAG_INFO;
# Y para localizar el alert_<SID>.log, revisa el valor de
# NAME='Default Trace File'.
  /u01/app/oracle/diag/rdbms/orclcdb/ORCLCDB/alert
# Vamos a poner un atajo que nos muestre las últimas líneas del log
# en la variable de entorno de esta base de datos.
vi ~/ENV/ORCLCDB.env
  alias check_alert='tail -f /u01/app/oracle/diag/rdbms/orclcdb/ORCLCDB/trace/alert_ORCLCDB.log'
  alias sysdba='sqlplus / as sysdba'
  # Ya que estamos, vamos a crear un par de tablespaces para empezar.
SQL> CREATE TABLESPACE LAB_D LOGGING DATAFILE '/u02/oradata/ORCLCDB/LAB_D.dbf' SIZE 1G AUTOEXTEND ON NEXT 10M MAXSIZE 5G;

  Tablespace created.

SQL> CREATE TABLESPACE LAB_I LOGGING DATAFILE '/u02/oradata/ORCLCDB/LAB_I.dbf' SIZE 1G AUTOEXTEND ON NEXT 10M MAXSIZE 2G;

  Tablespace created.
#--------------------------------------------------
# 2.6 ) EJECUCIÓN DE SCRIPTS
#--------------------------------------------------
# Necesitamos ejecutar una serie de scripts para crear vistas, PL/SQL,
# sinónimos del diccionario de datos y preparar el correcto
# funcionamiento de SQL*Plus.
SQL> @?/rdbms/admin/catalog.sql
SQL> @?/rdbms/admin/catproc.sql
SQL> @?/rdbms/admin/utlrp.sql
#--------------------------------------------------
# 2.7 ) OTRAS CONFIGURACIONES
#--------------------------------------------------
# Podemos, por ejemplo, hacer que la instancia se inicie de forma
# automática cuando el servidor se reinicie [3].
# También podemos configurar uno o varios servicios [6].
#
# Podemos reforzar la seguridad de la base de datos con distintas
# técnicas. Por ejemplo, usando perfiles con restricciones de
# contraseñas, roles y permisos según el usuario.
# También podemos encriptar los datafiles o tablaspaces completos con
# Transparent Data Encryption, para que no sean legibles si alguien
# intenta acceder a ellos desde el sistema operativo.
# Además, podemos usar Oracle Wallet como gestor de contraseñas seguro
# desde el lado del cliente, evitando tener credenciales en scripts o
# código de aplicación.
#
# Podemos instalar esquemas y datos dummy desde [7].
#--------------------------------------------------
# 3 ) BORRADO DE LA DB
#--------------------------------------------------
# Usamos el comando:
SQL> DROP DATABASE;
# Este comando borra todos los ficheros listados en el controlfile.
# Para que funcione correctamente, la base de datos debe estar montada
# en modo exclusivo y restrictivo.
