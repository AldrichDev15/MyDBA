#==================================================
# CREACIÓN DE DB 
#==================================================
#**************************************************
# 1) CONTEXTO Y ENTORNO 
#**************************************************
# Vamos a crear una DB en una máquina virtual Oracle
# Linux Server v8.10. Los recursos que le daremos a
# dicha máquina virtual serán los siguientes:
# - 3052M de memoria RAM.
# - 3 CPUs.
# - 200G de disco.
# Hemos instalado Oracle Database 19c en la máquina
# siguiendo los pasos del proceso de instalación oficial.
# Una vez lo tenemos instalado, procedemos a la creación
# de una DB.
# Para acceder a la VM, hemos configurado una regla NAT
# que redirige el puerto 2222 de nuestro host al ssh de 
# la VM (puerto 22). Por lo tanto, ejecutaremos
ssh -p 2222 oracle@localhost
#**************************************************
# 2) CREACIÓN DE UNA INSTANCIA MANUAL [1]
#**************************************************
# Para crear la instancia manual, usamos el comando 
"SQL> CREATE DATABASE"
# Vamos a ver su alcance. En este caso, NO vamos a 
# configurar ASM. 
# Para ver cómo se usa DBCA, ver [8] (no interactivo) y [9] (interactivo)
#--------------------------------------------------
# 2.1 ) ESTABLECER ORACLE_SID
#--------------------------------------------------
# El primer paso que se indica es elegir un nombre para
# la variable ORACLE_SID. En nuestro caso, se llamará
# ORCLCDB.
# Vamos a crear un directorio ENV en el home del usuario 
# oracle y en él, añadiremos los archivos de entorno para
# DB que creeemos.
mkdir -p ~/ENV
vi ~/ENV/ORCLCDB.env
 # DB PARAMS
 export ORACLE_SID=ORCLCDB

 # OTHERS
 export PS1="\u[$ORACLE_SID] \w>"
#
#--------------------------------------------------
# 2.2 ) MÉTODO DE AUTENTICACIÓN [2]
#--------------------------------------------------
# El siguiente paso trata sobre los métodos de autenticación.
# En nuestro caso, siempre vamos a poder entrar desde el
# OS como usuario oracle (quién ya tiene privilegio de SYSDBA),
# pero vamos a crear un fichero de contraseña aún así.
# Utilizaremos la utilidad ORAPWD. Este comando crea y 
# mantiene un archivo de contraseñas.
orapwd file=$ORACLE_HOME/dbs/orapw$ORACLE_SID \
    FORCE=y \
    FORMAT=12.2 \
    SYS=password \
# 
#--------------------------------------------------
# 2.3 ) CREACIÓN DEL PFILE
#--------------------------------------------------
# Estamos listos para crear el PFILE de nuestra DB. Creamos
# un PFILE porque aún no podemos generar un SPFILE al no 
# tener la DB creada.
# El fichero estará en $ORACLE_HOME/dbs/init$ORACLE_SID.ora.
vi $ORACLE_HOME/dbs/initORCLCDB.ora

 db_name='ORCLCDB'
 memory_target=1G
 processes = 150
 db_block_size=8192
 db_domain=''
 db_recovery_file_dest='/u03/fast_recovery_area'
 db_recovery_file_dest_size=30G
 diagnostic_dest='/u01/app/oracle'
 dispatchers='(PROTOCOL=TCP) (SERVICE=SRVORCLCDB)'
 open_cursors=300
 enable_pluggable_database=true
 remote_login_passwordfile='EXCLUSIVE'
 undo_tablespace='UNDOTBS1'
 # You may want to ensure that control files are created on separate physical
 # devices
 control_files = (/u01/ctrlf/control01.ctl, /u02/ctrlf/control02.ctl)
 compatible ='12.0.0'
#
# Este PFILE es mínimo.
# En el PFILE, la gestión del UNDO es automática por defecto
# pero podemos hacerla explícita añadiendo UNDO_MANAGEMENT=AUTO.
# Con este modo, el UNDO se guarda en el tablespace de UNDO y
# lo gestiona el motor de DB.
# 
# Si añadimos alguno de los siguientes parámetros en el PFILE:
# DB_CREATE_FILE_DEST, DB_CREATE_ONLINE_LOG_DEST_n o
# DB_RECOVERY_FILE_DEST, le decimos a la DB que cree y gestione 
# los ficheros para las estructuras de DB especificadas.
# 
# oracle recomienda usar Fast Recovery Area.
#--------------------------------------------------
# 2.4 ) CONEXIÓN A LA INSTANCIA Y SPFILE
#--------------------------------------------------
# Con las variables de entorno correctamente asignadas
su - oracle
source ~/ENV/ORCLCDB.env
sqlplus /nolog
SQL> CONNECT / AS SYSDBA

 Connected to an idle instance.
#
# Creamos un SPFILE con el PFILE que teníamos.
# Se crea en la ruta por defecto "$ORACLE_HOME/dbs/spfileORCLCDB.ora".
# 
SQL> CREATE SPFILE FROM PFILE='/u01/app/oracle/product/19.0.0/dbhome_1/dbs/initORCLCDB.ora';

 File created.
# 
# Existen dos tipos de parámetros de inicio:
# - Dinámicos: Pueden cambiarse en la propia instancia y son immediatos.
# - Estáticos: Se requiere reinicio da la DB.
# Para cambiarlos se usa el comando ALTER SYSTEM SET y tiene varios alcances
# (SCOPES)
# - SPFILE: Cambia en el SPFILE solo y no se hace ningún cambio en la instancia 
#   actual.
# - MEMORY: Se cambia el parámetro dinámico para esta instancia, pero no poersistirá
#   en futuras
# - BOTH: Combina las anteriores. Solo sirve para parámetros dinámicos.
# Si usamos SPFILE o BOTH hay que inicilizar la DB con un SPFILE. Si lo hemos 
# hecho así, SCOPE=BOTH será el comportamiento por defecto.
# Si hemos usado un PFILE será MEMORY.
# También podemos usar ALTER SYSTEM RESET para dejar el parámetro con su valor por defecto.
# CUIDADO: El comportamiento de este comando cambia según la DB sea CDB o PDB.
# Si usamos RMAN, podemos hacer un backup solo del SPFILE, aunque,
# si hacemos un backup de la DB estará incluido.
# Algunos métodos para ver los parámetros son:
SQL> SHOW PARAMETERS --Sesión actual
SQL> SHOW SPPARAMETERS --Los del SPFILE
SQL> CREATE PFILE 
SQL> SELECT * FROM V$PARAMETER2; --Sesión actual
SQL> SELECT * FROM V$SYSTEM_PARAMETER2; --Instancia actual
SQL> SELECT * FROM V$SPPARAMETER; --Contenido actual del SPFILE
#
SQL> SELECT INSTANCE_NAME,DATABASE_STATUS,STATUS FROM V$INSTANCE;

 INSTANCE_NAME    DATABASE_STATUS   STATUS
 ---------------- ----------------- ------------
 ORCLCDB          ACTIVE            STARTED

# En este punto la memoria está asignada y los procesos 
# levantados, pero la DB en sí no existe.
#--------------------------------------------------
# 2.5 ) CREACIÓN DE LA DB
#--------------------------------------------------
CREATE DATABASE ORCLCDB
   USER SYS IDENTIFIED BY "0R4C|34DM1N"
   USER SYSTEM IDENTIFIED BY "ORASYSTEM"
   LOGFILE GROUP 1 ('/u03/online_redo/redo01a.log') SIZE 100M BLOCKSIZE 512,
           GROUP 2 ('/u03/online_redo/redo02a.log') SIZE 100M BLOCKSIZE 512,
           GROUP 3 ('/u03/online_redo/redo03a.log') SIZE 100M BLOCKSIZE 512
   MAXLOGHISTORY 100
   MAXLOGFILES 16
   MAXLOGMEMBERS 3
   MAXDATAFILES 1024
   CHARACTER SET AL32UTF8
   NATIONAL CHARACTER SET AL16UTF16
   EXTENT MANAGEMENT LOCAL
   DATAFILE '/u02/oradata/ORCLCDB/system01.dbf'
     SIZE 700M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE 15G
   SYSAUX DATAFILE '/u02/oradata/ORCLCDB/sysaux01.dbf'
     SIZE 550M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE 15G
   DEFAULT TABLESPACE USERS
      DATAFILE '/u02/oradata/ORCLCDB/users01.dbf'
      SIZE 500M REUSE AUTOEXTEND ON NEXT 100M MAXSIZE 10G
   DEFAULT TEMPORARY TABLESPACE TEMPTS1
      TEMPFILE '/u02/oradata/ORCLCDB/temp01.dbf'
      SIZE 200M REUSE AUTOEXTEND ON NEXT 640K MAXSIZE 15G
   UNDO TABLESPACE UNDOTBS1
      DATAFILE '/u02/oradata/ORCLCDB/undotbs01.dbf'
      SIZE 200M REUSE AUTOEXTEND ON NEXT 5120K MAXSIZE 15G
    ENABLE PLUGGABLE DATABASE
      SEED
      FILE_NAME_CONVERT =
      ('/u02/oradata/ORCLCDB/',
       '/u02/oradata/ORCLCDB/pdbseed/');

  Database created.

# CUIDADO: Las contraseñas son case-sensitive.
# 
# CUIDADO: Este comando NO crea directorios, deben 
# existir previamente. Si el comando fallase, debemos 
# borrar todo lo que haya creado la anterior ejecución.
# 
# CUIDADO: Si vamos a usar OMF, vigilar los parámetros 
# por defecto del CREATE DATABASE.
#
# NOTA: EXTENT MANAGEMENT LOCAL crea un bitmap en cada 
# datafile para manejar los extents. Esta es la opción 
# recomendada, pero NO es la opción por defecto, por tanto
# hay que especificarlo.
#
# NOTA: El tablespace SYSAUX es un tablespace auxiliar al 
# de SYSTEM, obligatorio y que sirve como tablespace por defecto
# para muchas funcionalidades y productos que antes requerían de 
# uno propio. También reduce la carga en el tablespace de SYSTEM.
# podemos usar V$SYSAUX_OCCUPANTS para ver los componentes que
# hay [4]. 
#
# NOTA: En el CREATE DATABASE podemos elegir si los
# tablespaces que crearemos por defecto son bigfile o smallfile.
# Podemos consultar el tipo de tablespace por defecto 
# (smallfile o bigfile) con
SQL> SELECT PROPERTY_VALUE FROM DATABASE_PROPERTIES 
WHERE PROPERTY_NAME = 'DEFAULT_TBS_TYPE';
# Por defecto es smallfile. 
# También podemos alterar el comportamiento por defecto especificando
# nosotros la opción, excepto con los de SYSTEM y SYSAUX.
#
# NOTA: Si tenemos UNDO_MANAGEMENT en AUTO y no especificamos 
# el tablespace de UNDO, se nos creará uno por defecto llamado SYS_UNDOTBS.
#
# NOTA: Oracle recomienda crear un tablespace por defecto para los
# usuarios no-SYSTEM. Si no lo hacemos, el tablespace por defecto será SYSTEM.
# Lo mismo ocurre con el tablespace temporal. Si no creamos uno por defecto
# y no especificamos uno al crear el usuario, el tablespace temporal por 
# defecto es SYSTEM (si no está en locally managed [5]), lo que es una mala práctica.
# Lo mismo ocurre con undo, si no creamos ningún tablespace undo y NO tenemos 
# el parámetro UNDO_TABLESPACE en AUTO, los datos de undo se escriben en SYSTEM.
#
# Cuando ejecutamos un "CREATE DATABASE" hacemos al menos las 
# siguientes operaciones:
# - Crear los datafiles, controlfiles y los online redologs.
#   (y establecer el modo ARCHIVELOG)
# - Crear los tablespaces de SYSTEM y SYSAUX.
# - Crear el diccionario de datos.
# - Establecer el charset y la zona horaria.
#   La zona horaria por defecto es la del host y se puede cambiar para ciertas 
#   sesiones. Para ver las zonas horarias válidas, ver:
#   "$ORACLE_HOME/oracore/zoneinfo/timezlrg_11.dat" (el que se usa por defecto) 
#   "ORACLE_HOME/oracore/zoneinfo/timezone_11.dat" (contiene menos zonas)
#   o en SQL*Plus ejecutar
SQL> SELECT * FROM V$TIMEZONE_NAMES;
# - Montar y abrir la DB.
#
# NOTA: Para ver las trazas, ejecutar
SQL> SELECT * FROM V$DIAG_INFO;
# Y para el alert_<SID>.log, fijarnos en el NAME='Default Trace File'.
  /u01/app/oracle/diag/rdbms/orclcdb/ORCLCDB/alert
# Vamos a poner un atajo que nos muestre las últimas líneas
# de este fichero en las variables de entorno de esta DB.
vi ~/ENV/ORCLCDB.env
  alias check_alert='tail -f /u01/app/oracle/diag/rdbms/orclcdb/ORCLCDB/trace/alert_ORCLCDB.log'
  alias sysdba='sqlplus / as sysdba'
  # Ya que estamos, vamos a crear un par de tablespaces para empezar.
SQL> CREATE TABLESPACE LAB_D LOGGING DATAFILE '/u02/oradata/ORCLCDB/LAB_D.dbf' SIZE 1G AUTOEXTEND ON NEXT 10M MAXSIZE 5G;
  
  Tablespace created.

SQL> CREATE TABLESPACE LAB_I LOGGING DATAFILE '/u02/oradata/ORCLCDB/LAB_I.dbf' SIZE 1G AUTOEXTEND ON NEXT 10M MAXSIZE 2G;
  
  Tablespace created.

# Podemos configurar también uno o varios servicios [6].
# 
# Podemos configurar mejor la seguridad de nuestra DB una vez creada con distintas técnicas.
# Podemos usar perfiles que tengan restricciones en cuanto a contraseñas, asignar roles y permisos

# Además, podemos usar Oracle wallet como gestor de contraseñas seguro desde el lado del cliente. Así
# evitamos tenerlo en scripts o código de aplicaciones.
